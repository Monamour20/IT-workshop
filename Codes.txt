Exercise 5: User-defined function FtoC and thermal expansion problem
% ---- file FtoC.m (function M-file) ----
function C = FtoC(F)
    C = (F - 32) * 5 / 9;
end
% ---- script exp6_ex5.m (calls the function) ----
alpha = 23e-6;
L = 4.5;   W = 2.25;
C1 = FtoC(40);   C2 = FtoC(92);
deltaT = C2 - C1;
 
deltaL = alpha * L * deltaT;
deltaW = alpha * W * deltaT;
deltaArea = (L + deltaL) * (W + deltaW) - (L * W);
 
disp('Temperature change (C)');  disp(deltaT);
disp('Change in Area (m^2)');     disp(deltaArea);

Exercise 4: Matrix built with nested loops

m = input('Enter number of columns: ');
A = zeros(n, m);
 
for j = 1:m,  A(1,j) = j;  end
for i = 1:n,  A(i,1) = i;  end
for i = 2:n
    for j = 2:m
        A(i,j) = A(i-1,j) + A(i,j-1);

Exercise 5: Retirement savings
balance = 300000;  interest = 0.05;  withdraw = 25000;  inflation = 0.02;
year = 0;  withdrawal = [];  balanceHistory = [];
 
while balance > 0
    year = year + 1;
    balance = balance * (1 + interest);
    balance = balance - withdraw;
    withdrawal(year) = withdraw;
    balanceHistory(year) = max(balance, 0);
    withdraw = withdraw * (1 + inflation);
end
disp(['Money lasts for ', num2str(year), ' years']);
 
subplot(2,1,1); plot(1:year, withdrawal, 'b', 'LineWidth', 2);
title('Yearly Withdrawals'); xlabel('Year'); ylabel('Withdrawal'); grid on;
subplot(2,1,2); plot(1:year, balanceHistory, 'r', 'LineWidth', 2);
title('Account Balance'); xlabel('Year'); ylabel('Balance'); grid on;
